PART 3 — Matplotlib (Visualization)

What is Matplotlib?

Matplotlib is Python's most popular graphing/charting library. The exam specifically asks for visualization graphs.

import matplotlib.pyplot as plt    # always as plt — standard

The Golden Template

import matplotlib.pyplot as plt

plt.figure(figsize=(8, 5))
plt.plot([1, 2, 3], [4, 5, 6])
plt.title("My Chart Title")
plt.xlabel("X Axis Label")
plt.ylabel("Y Axis Label")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

Tip: Always end with plt.show().

Line Plot

import matplotlib.pyplot as plt

months = [1, 2, 3, 4, 5, 6]
sales =  [200, 350, 300, 450, 500, 420]

plt.figure(figsize=(8, 4))
plt.plot(months, sales, color="blue", marker="o", linestyle="--", linewidth="2")
plt.title("Monthly Sales")
plt.xlabel("Month")
plt.ylabel("Sales")
plt.grid(True)
plt.show()

Line styles:

linestyle="-"   # solid
linestyle="--"  # dashed
linestyle=":"   # dotted
linestyle="-."  # dash-dot

Marker options:

marker="o"   # circle
marker="s"   # square
marker="^"   # triangle
marker="*"   # star

Bar Chart

import matplotlib.pyplot as plt

subjects = ["Math", "Physics", "Python", "English", "Stats"]
marks    = [85, 72, 95, 88, 79]

plt.figure(figsize=(8, 5))
plt.bar(subjects, marks, color="steelblue", edgecolor="black")
plt.title("Student Marks by Subject")
plt.xlabel("Subject")
plt.ylabel("Marks")
plt.ylim(0, 100)
plt.show()

Horizontal Bar Chart

plt.barh(subjects, marks, color="coral")

Histogram

import matplotlib.pyplot as plt
import numpy as np

scores = np.random.randint(40, 100, 50)

plt.figure(figsize=(8, 5))
plt.hist(scores, bins=10, color="green", edgecolor="black")
plt.title("Score Distribution")
plt.xlabel("Score")
plt.ylabel("Frequency")
plt.show()

Note: bins divides the data into groups.

Scatter Plot

import matplotlib.pyplot as plt
import numpy as np

hours_studied = [2, 3, 4, 5, 6, 7, 8, 9]
marks_obtained = [45, 55, 60, 70, 75, 85, 88, 95]

plt.figure(figsize=(7, 5))
plt.scatter(hours_studied, marks_obtained, color="red", s=100, marker="o")
plt.title("Study Hours vs Marks")
plt.xlabel("Hours Studied")
plt.ylabel("Marks")
plt.grid(True)
plt.show()

Pie Chart

import matplotlib.pyplot as plt

categories = ["Python", "Java", "C++", "JavaScript"]
values     = [40, 25, 20, 15]
colors     = ["gold", "lightblue", "lightgreen", "coral"]
explode    = [0.1, 0, 0, 0]

plt.figure(figsize=(7, 7))
plt.pie(values,
        labels=categories,
        colors=colors,
        explode=explode,
        autopct="%1.1f%%",
        shadow=True)
plt.title("Programming Language Popularity")
plt.show()

Multiple Plots with subplot()

import matplotlib.pyplot as plt

x = [1, 2, 3, 4, 5]
y1 = [1, 4, 9, 16, 25]
y2 = [1, 2, 3, 4, 5]

plt.figure(figsize=(12, 4))
plt.subplot(1, 2, 1)
plt.plot(x, y1, color="blue")
plt.title("Quadratic")

plt.subplot(1, 2, 2)
plt.plot(x, y2, color="red")
plt.title("Linear")

plt.tight_layout()
plt.show()

Tip: position starts from 1.

Customization Cheatsheet

# Colors
color="red"
color="#FF5733"
color=(0.1, 0.5, 0.9)

# Multiple lines
plt.plot(x, y1, label="Line 1")
plt.plot(x, y2, label="Line 2")
plt.legend()

# Axis limits
plt.xlim(0, 10)
plt.ylim(0, 100)

# Tick marks
plt.xticks([0, 2, 4, 6, 8, 10])
plt.yticks([0, 25, 50, 75, 100])

# Text on plot
plt.text(x=3, y=50, s="Important Point", fontsize=12)

# Save plot
plt.savefig("chart.png", dpi=300, bbox_inches="tight")

Pandas + Matplotlib Together

import pandas as pd
import matplotlib.pyplot as plt

data = {
    "Month": ["Jan", "Feb", "Mar", "Apr", "May"],
    "Sales": [1200, 1500, 1100, 1800, 2000]
}
df = pd.DataFrame(data)

plt.figure(figsize=(8, 5))
plt.plot(df["Month"], df["Sales"], marker="o", color="blue")
plt.title("Monthly Sales Report")
plt.xlabel("Month")
plt.ylabel("Sales (PKR)")
plt.grid(True)
plt.show()

Bar chart from DataFrame

plt.bar(df["Month"], df["Sales"], color="steelblue")
plt.title("Sales per Month")
plt.show()

Practice Programs — Visualization

import pandas as pd
import matplotlib.pyplot as plt

data = {
    "Name":   ["Ali", "Sara", "Zain", "Hira", "Omer"],
    "Marks":  [85, 92, 78, 95, 70],
    "Age":    [20, 21, 22, 20, 23],
    "City":   ["Karachi", "Lahore", "Karachi", "Islamabad", "Lahore"]
}
df = pd.DataFrame(data)

print(df.describe())
print(df["Marks"].mean())
print(df.groupby("City")["Marks"].mean())

fig, axes = plt.subplots(1, 3, figsize=(15, 4))
axes[0].bar(df["Name"], df["Marks"], color="steelblue")
axes[0].set_title("Marks per Student")
axes[0].set_xlabel("Student")
axes[0].set_ylabel("Marks")
axes[1].hist(df["Marks"], bins=5, color="green", edgecolor="black")
axes[1].set_title("Marks Distribution")
axes[2].scatter(df["Age"], df["Marks"], color="red", s=100)
axes[2].set_title("Age vs Marks")
axes[2].set_xlabel("Age")
axes[2].set_ylabel("Marks")
plt.tight_layout()
plt.show()

Short Answer Theory

What is Matplotlib? A Python library for building charts and graphs.

Why use plt.show()? It displays the plot window.

When to use a histogram vs bar chart? Histogram for distribution of numeric data, bar chart for category comparisons.

What does axis=0 vs axis=1 mean? axis=0 is column-wise. axis=1 is row-wise.
